# https://www.macgly.com/robots.txt
# This file is the single source of truth. Vercel serves everything in public/
# before it applies any rewrite, so the /robots.txt rewrite in vercel.json never
# fires — do not add rules to the API instead of here.

User-agent: *
Allow: /

# Private / authenticated / transactional pages
Disallow: /dashboard/
Disallow: /checkout
Disallow: /cart
Disallow: /order-confirmation/
Disallow: /login
Disallow: /register
Disallow: /forgot-password
Disallow: /reset-password/
Disallow: /api/

# /track-order?id=...&phone=... puts an order id and a customer phone number in
# the URL. It must never be crawled or indexed.
Disallow: /track-order?

# Faceted and paginated catalogue URLs. Every product, category and blog post
# has its own clean URL in the sitemap, so blocking these loses no coverage —
# it only stops the same 500 products being crawled under thousands of filter
# permutations.
# Two rules per parameter: "?name=" matches it as the first parameter,
# "&name=" matches it anywhere after. One rule alone would miss half the URLs.
Disallow: /products?
Disallow: /*?page=
Disallow: /*&page=
Disallow: /*?sort=
Disallow: /*&sort=
Disallow: /*?search=
Disallow: /*&search=
Disallow: /*?minPrice=
Disallow: /*&minPrice=
Disallow: /*?maxPrice=
Disallow: /*&maxPrice=
Disallow: /*?minRating=
Disallow: /*&minRating=
Disallow: /*?inStock=
Disallow: /*&inStock=
Disallow: /*?brand=
Disallow: /*&brand=
Disallow: /*?featured=
Disallow: /*&featured=

# Affiliate links are the canonical URL plus ?ref=CODE, and App.jsx reads that
# parameter on every route — so an unblocked ?ref= duplicates the entire site
# once per affiliate.
Disallow: /*?ref=
Disallow: /*&ref=

# The favicon, the Open Graph card and the icons must stay crawlable. Google
# will not show a favicon in search results if it cannot fetch the file.
Allow: /favicon.ico
Allow: /favicon.svg
Allow: /favicon.png
Allow: /favicon-192.png
Allow: /favicon-512.png
Allow: /apple-touch-icon.png
Allow: /og-image.png

Sitemap: https://www.macgly.com/sitemap.xml
